cmake_minimum_required(VERSION 3.16)
project(EduDB VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
option(EDUDB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(EDUDB_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(edudb_core src/storage.cpp src/aries_logical.cpp src/query_reuse.cpp src/tree_control.cpp src/mvto_control.cpp src/input_limits.cpp src/btree.cpp src/sql.cpp src/access.cpp src/io.cpp src/interchange.cpp src/query_spill.cpp src/query_operators.cpp src/transaction_control.cpp src/managed_block_device.cpp src/buffer_pool.cpp src/lock_manager.cpp src/execution_progress.cpp src/record_codec.cpp src/query_relational.cpp src/aries_recovery.cpp src/page_version_graph.cpp src/cluster_heap.cpp src/materialized_delta.cpp src/query_semijoin.cpp src/query_topk.cpp src/format_copy.cpp src/format_migrate.cpp src/recovery_archive.cpp src/fencing.cpp src/standby_transport.cpp)
target_include_directories(edudb_core PUBLIC include third_party)
target_link_libraries(edudb_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(edudb_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(edudb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(edudb src/main.cpp)
target_link_libraries(edudb PRIVATE edudb_core)
if(WIN32)
  target_link_libraries(edudb_core PUBLIC crypt32 ws2_32)
  target_compile_definitions(edudb_core PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(edudb PRIVATE ws2_32 crypt32 shell32)
  if(MINGW AND NOT EDUDB_SANITIZERS)
    target_link_options(edudb PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
enable_testing()
foreach(suite buffer_lifetime_recovery io_owner_exit protocol_recovery_matrix shared_scan_sql aries_logical_reservation logical_fence aries_logical_sql query_reuse query_self_join_sql tree write_scope mvto standby_input input_limits parser_input input_limits_sql mv2pl query_semijoin query_decorrelation_sql query_topk query_topk_sql aries_commit_safety concurrency_composition_sql range_lock range_sql range_snapshot btree storage sql access sql_ch11 ch11_storage query_spill query_optimizer transaction_protocols ch10_storage buffer_pool managed_storage managed_media lock_manager record_codec record_storage query_relational query_relational_sql query_subquery_sql change_feed aries aries_sql page_version_graph page_version_graph_sql cluster_storage cluster_sql cluster_aries_sql materialized_delta query_exact_aggregate_sql query_materialized_sql query_materialized_delta_sql query_materialized_rewrite_sql materialized_cluster_sql backup_sql recovery_archive fencing io_owner recovery_product_sql format_copy format4)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_test.cpp")
    add_executable(${suite}_test tests/${suite}_test.cpp)
    target_link_libraries(${suite}_test PRIVATE edudb_core)
    if(WIN32 AND (suite STREQUAL "managed_media" OR suite STREQUAL "aries_sql"))
      target_link_libraries(${suite}_test PRIVATE shell32)
    endif()
    add_test(NAME ${suite} COMMAND ${suite}_test)
    set_tests_properties(${suite} PROPERTIES TIMEOUT 300)
    if(MINGW AND NOT EDUDB_SANITIZERS)
      target_link_options(${suite}_test PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
endforeach()

    target_link_options(${driver} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endforeach()
